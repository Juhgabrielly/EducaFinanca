USE educafinanca;

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    email VARCHAR(150) NOT NULL UNIQUE,
    senha VARCHAR(255),
    tipo ENUM('USUARIO', 'ADMIN') DEFAULT 'USUARIO',
    plano ENUM('FREE', 'PREMIUM') DEFAULT 'FREE',
    status BOOLEAN DEFAULT TRUE,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE assinaturas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    plano ENUM('FREE', 'PREMIUM') NOT NULL,
    status ENUM('ATIVA', 'EXPIRADA', 'CANCELADA') DEFAULT 'ATIVA',
    inicio DATETIME DEFAULT CURRENT_TIMESTAMP,
    fim DATETIME,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);


CREATE TABLE cursos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    descricao TEXT,
    conteudo TEXT,
    exclusivo BOOLEAN DEFAULT FALSE,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE progresso_cursos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    curso_id INT NOT NULL,
    progresso DECIMAL(5,2) DEFAULT 0,
    concluido BOOLEAN DEFAULT FALSE,

    UNIQUE (usuario_id, curso_id),

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (curso_id) REFERENCES cursos(id)
);

CREATE TABLE quizzes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    curso_id INT,
    titulo VARCHAR(150) NOT NULL,
    descricao TEXT,
    exclusivo BOOLEAN DEFAULT FALSE,

    FOREIGN KEY (curso_id) REFERENCES cursos(id)
);

CREATE TABLE questoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    quiz_id INT NOT NULL,
    pergunta TEXT NOT NULL,
    resposta_correta VARCHAR(255) NOT NULL,

    FOREIGN KEY (quiz_id) REFERENCES quizzes(id)
);

CREATE TABLE resultados_quiz (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    quiz_id INT NOT NULL,
    pontuacao DECIMAL(5,2) NOT NULL,
    realizado_em DATETIME DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (quiz_id) REFERENCES quizzes(id)
);

CREATE TABLE metas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    nome VARCHAR(150) NOT NULL,
    valor_meta DECIMAL(12,2) NOT NULL,
    valor_atual DECIMAL(12,2) DEFAULT 0,
    criada_em DATETIME DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE tipos_investimento (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao TEXT
);
CREATE TABLE investimentos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    tipo_investimento_id INT NOT NULL,
    nome VARCHAR(150) NOT NULL,
    valor_investido DECIMAL(12,2) NOT NULL,
    valor_atual DECIMAL(12,2) DEFAULT 0,
    rentabilidade DECIMAL(8,2) DEFAULT 0,
    criado_em DATETIME DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
    FOREIGN KEY (tipo_investimento_id) REFERENCES tipos_investimento(id)
);

CREATE TABLE reservas_emergencia (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL UNIQUE,
    valor DECIMAL(12,2) DEFAULT 0,
    atualizado_em DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE movimentacoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    tipo ENUM('RECEITA', 'DESPESA') NOT NULL,
    valor DECIMAL(12,2) NOT NULL,
    descricao VARCHAR(255),
    data_movimentacao DATE NOT NULL,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);
